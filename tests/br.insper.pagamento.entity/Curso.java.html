package br.insper.avaliacao.entity;

import br.insper.avaliacao.dto.CursoDTO;
import jakarta.persistence.*;
import lombok.AllArgsConstructor;
import lombok.Data;
import lombok.NoArgsConstructor;

import java.util.ArrayList;
import java.util.List;

@Entity
@Table(name = "avaliacaos")
@Data
@NoArgsConstructor
@AllArgsConstructor
public class Curso {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    private int quantidadeAlunos;

    private String nome;

    @ElementCollection(fetch = FetchType.EAGER)
    @CollectionTable(
            name = "curso_materias",
            joinColumns = @JoinColumn(name = "curso_id")
    )
    @Column(name = "materia")
    private List<String> materias = new ArrayList<>();

    private boolean deletado = false;

    public static Curso fromDTO(CursoDTO dto) {
        Curso avaliacao = new Curso();

        avaliacao.setNome(dto.getNome());
        avaliacao.setMaterias(dto.getMaterias());
        avaliacao.setQuantidadeAlunos(dto.getQuantidade_alunos());
        avaliacao.setDeletado(false);

        return avaliacao;
    }
}
